// ========================================================
// Constructor
// ========================================================

LensLinksMediaStream::LensLinksMediaStream(
    LensLinksMediaSource* source)
    : source_(source)
{
    if (source_ != nullptr)
    {
        source_->AddRef();
    }

    HRESULT hr =
        MFCreateEventQueue(
            &eventQueue_);

    if (FAILED(hr))
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "MFCreateEventQueue failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;
    }

    streamState_ =
        MF_STREAM_STATE_STOPPED;

    std::cout
        << "[LensLinksMediaStream] created"
        << std::endl;
}

// ========================================================
// Destructor
// ========================================================

LensLinksMediaStream::~LensLinksMediaStream()
{
    shutdown_ = true;

    if (eventQueue_ != nullptr)
    {
        eventQueue_->Shutdown();

        eventQueue_->Release();

        eventQueue_ = nullptr;
    }

    if (streamDescriptor_ != nullptr)
    {
        streamDescriptor_->Release();

        streamDescriptor_ = nullptr;
    }

    if (source_ != nullptr)
    {
        source_->Release();

        source_ = nullptr;
    }

    std::cout
        << "[LensLinksMediaStream] destroyed"
        << std::endl;
}

// ========================================================
// IUnknown
// ========================================================

HRESULT STDMETHODCALLTYPE
LensLinksMediaStream::QueryInterface(
    REFIID riid,
    void** ppvObject)
{
    if (ppvObject == nullptr)
    {
        return E_POINTER;
    }

    *ppvObject = nullptr;

    if (riid == IID_IUnknown)
    {
        *ppvObject =
            static_cast<IUnknown*>(
                static_cast<IMFMediaStream*>(this));
    }
    else if (riid == IID_IMFMediaEventGenerator)
    {
        *ppvObject =
            static_cast<IMFMediaEventGenerator*>(
                static_cast<IMFMediaStream*>(this));
    }
    else if (riid == IID_IMFMediaStream)
    {
        *ppvObject =
            static_cast<IMFMediaStream*>(this);
    }
    else if (riid == IID_IMFMediaStream2)
    {
        *ppvObject =
            static_cast<IMFMediaStream2*>(this);
    }
    else
    {
        return E_NOINTERFACE;
    }

    AddRef();

    return S_OK;
}

ULONG STDMETHODCALLTYPE
LensLinksMediaStream::AddRef()
{
    return static_cast<ULONG>(
        InterlockedIncrement(
            &refCount_));
}

ULONG STDMETHODCALLTYPE
LensLinksMediaStream::Release()
{
    ULONG count =
        static_cast<ULONG>(
            InterlockedDecrement(
                &refCount_));

    if (count == 0)
    {
        delete this;
    }

    return count;
}

// IMFMediaEventGenerator

HRESULT STDMETHODCALLTYPE
LensLinksMediaStream::GetEvent(
    DWORD dwFlags,
    IMFMediaEvent** ppEvent)
{
    if (ppEvent == nullptr)
    {
        return E_POINTER;
    }

    *ppEvent = nullptr;

    if (shutdown_)

        return MF_E_SHUTDOWN;

    {

    }

    return eventQueue_->GetEvent(
        dwFlags,
        ppEvent);
}

// ========================================================
// IMFMediaStream2
// ========================================================

HRESULT STDMETHODCALLTYPE
LensLinksMediaStream::SetStreamState(
    MF_STREAM_STATE state)
{
    if (shutdown_)
    {
        return MF_E_SHUTDOWN;
    }

    switch (state)
    {
        case MF_STREAM_STATE_STOPPED:
        case MF_STREAM_STATE_RUNNING:
        case MF_STREAM_STATE_PAUSED:
            break;

        default:
            return E_INVALIDARG;
    }

    std::cout
        << "[LensLinksMediaStream] "
        << "SetStreamState "
        << static_cast<int>(state)
        << std::endl;

    streamState_ = state;

    return S_OK;
}

HRESULT STDMETHODCALLTYPE
LensLinksMediaStream::GetStreamState(
    MF_STREAM_STATE* pState)
{
    if (pState == nullptr)
    {
        return E_POINTER;
    }

    *pState =
        MF_STREAM_STATE_STOPPED;

    if (shutdown_)
    {
        return MF_E_SHUTDOWN;
    }

    *pState =
        streamState_;

    return S_OK;
}

// ========================================================
// RequestSample
// ========================================================

HRESULT STDMETHODCALLTYPE
LensLinksMediaStream::RequestSample(
    IUnknown* pToken)
{
    if (shutdown_)
    {
        return MF_E_SHUTDOWN;
    }

    if (eventQueue_ == nullptr)
    {
        return MF_E_NOT_INITIALIZED;
    }

    if (streamDescriptor_ == nullptr)
    {
        return MF_E_NOT_INITIALIZED;
    }

    // ----------------------------------------------------
    // Make sure stream is running
    // ----------------------------------------------------

    if (streamState_ != MF_STREAM_STATE_RUNNING)
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample ignored: stream not running"
            << std::endl;

        return MF_E_NOT_INITIALIZED;
    }

    // ----------------------------------------------------
    // Obtain camera frame
    // ----------------------------------------------------

    CameraFrame frame;

    if (!VirtualCamera::instance().getFrame(frame))
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: no frame available"
            << std::endl;

        return MF_E_NOT_INITIALIZED;
    }

    // ----------------------------------------------------
    // Validate dimensions
    // ----------------------------------------------------

    if (frame.width <= 0 ||
        frame.height <= 0)
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: invalid dimensions "
            << frame.width
            << "x"
            << frame.height
            << std::endl;

        return MF_E_INVALIDMEDIATYPE;
    }

    if (frame.pixels.empty())
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: empty frame"
            << std::endl;

        return MF_E_NOT_INITIALIZED;
    }

    // ----------------------------------------------------
    // RGB32
    // ----------------------------------------------------

    constexpr size_t BYTES_PER_PIXEL = 4;

    const size_t width =
        static_cast<size_t>(
            frame.width);

    const size_t height =
        static_cast<size_t>(
            frame.height);

    if (height != 0 &&
        width >
            (std::numeric_limits<size_t>::max)() /
                height)
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: dimension overflow"
            << std::endl;

        return E_INVALIDARG;
    }

    const size_t pixelCount =
        width * height;

    if (pixelCount >
        (std::numeric_limits<size_t>::max)() /
            BYTES_PER_PIXEL)
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: pixel buffer overflow"
            << std::endl;

        return E_INVALIDARG;
    }

    const size_t expectedSize =
        pixelCount *
        BYTES_PER_PIXEL;

    if (frame.pixels.size() != expectedSize)
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: invalid RGB32 buffer size "
            << "expected="
            << expectedSize
            << " actual="
            << frame.pixels.size()
            << std::endl;

        return MF_E_INVALIDMEDIATYPE;
    }

    if (expectedSize >
        static_cast<size_t>(
            (std::numeric_limits<DWORD>::max)()))
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "RequestSample: buffer too large"
            << std::endl;

        return E_INVALIDARG;
    }

    const DWORD bufferSize =
        static_cast<DWORD>(
            expectedSize);

    // ----------------------------------------------------
    // Create sample
    // ----------------------------------------------------

    IMFSample* sample = nullptr;

    HRESULT hr =
        MFCreateSample(
            &sample);

    if (FAILED(hr))
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "MFCreateSample failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;

        return hr;
    }

    // ----------------------------------------------------
    // Create media buffer
    // ----------------------------------------------------

    IMFMediaBuffer* buffer = nullptr;

    hr =
        MFCreateMemoryBuffer(
            bufferSize,
            &buffer);

    if (FAILED(hr))
    {
        sample->Release();

        std::cout
            << "[LensLinksMediaStream] "
            << "MFCreateMemoryBuffer failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;

        return hr;
    }

    // ----------------------------------------------------
    // Copy frame into buffer
    // ----------------------------------------------------

    BYTE* destination = nullptr;
    DWORD maxLength = 0;
    DWORD currentLength = 0;

    hr =
        buffer->Lock(
            &destination,
            &maxLength,
            &currentLength);

    if (SUCCEEDED(hr))
    {
        if (destination == nullptr)
        {
            hr = E_POINTER;
        }
        else if (maxLength < bufferSize)
        {
            hr = E_OUTOFMEMORY;
        }
        else
        {
            std::memcpy(
                destination,
                frame.pixels.data(),
                bufferSize);

            hr =
                buffer->SetCurrentLength(
                    bufferSize);
        }

        buffer->Unlock();
    }

    if (FAILED(hr))
    {
        buffer->Release();
        sample->Release();

        std::cout
            << "[LensLinksMediaStream] "
            << "buffer copy failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;

        return hr;
    }

    // ----------------------------------------------------
    // Attach buffer to sample
    // ----------------------------------------------------

    hr =
        sample->AddBuffer(
            buffer);

    buffer->Release();

    if (FAILED(hr))
    {
        sample->Release();

        std::cout
            << "[LensLinksMediaStream] "
            << "AddBuffer failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;

        return hr;
    }

    // ----------------------------------------------------
    // Preserve request token
    // ----------------------------------------------------

    if (pToken != nullptr)
    {
        hr =
            sample->SetUnknown(
                MFSampleExtension_Token,
                pToken);

        if (FAILED(hr))
        {
            std::cout
                << "[LensLinksMediaStream] "
                << "SetUnknown token failed hr=0x"
                << std::hex
                << static_cast<unsigned long>(hr)
                << std::dec
                << std::endl;

            sample->Release();

            return hr;
        }
    }

    // ----------------------------------------------------
    // Frame timing
    // ----------------------------------------------------

    static LONGLONG frameNumber = 0;

    constexpr LONGLONG FRAME_RATE = 30;

    constexpr LONGLONG FRAME_DURATION =
        10'000'000LL / FRAME_RATE;

    const LONGLONG sampleTime =
        frameNumber *
        FRAME_DURATION;

    ++frameNumber;

    hr =
        sample->SetSampleTime(
            sampleTime);

    if (FAILED(hr))
    {
        sample->Release();

        return hr;
    }

    hr =
        sample->SetSampleDuration(
            FRAME_DURATION);

    if (FAILED(hr))
    {
        sample->Release();

        return hr;
    }

    // ----------------------------------------------------
    // Queue MEMediaSample
    // ----------------------------------------------------

    PROPVARIANT value;

    PropVariantInit(
        &value);

    value.vt =
        VT_UNKNOWN;

    value.punkVal =
        sample;

    sample->AddRef();

    hr =
        eventQueue_->QueueEventParamVar(
            MEMediaSample,
            GUID_NULL,
            S_OK,
            &value);

    PropVariantClear(
        &value);

    // Release our original sample reference.
    sample->Release();

    if (FAILED(hr))
    {
        std::cout
            << "[LensLinksMediaStream] "
            << "Queue MEMediaSample failed hr=0x"
            << std::hex
            << static_cast<unsigned long>(hr)
            << std::dec
            << std::endl;

        return hr;
    }

    std::cout
        << "[LensLinksMediaStream] "
        << "MEMediaSample queued "
        << frame.width
        << "x"
        << frame.height
        << " RGB32 "
        << bufferSize
        << " bytes"
        << std::endl;

    return S_OK;
}

// ========================================================
// Stream descriptor
// ========================================================

HRESULT
LensLinksMediaStream::setStreamDescriptor(
    IMFStreamDescriptor* descriptor)
{
    if (descriptor == nullptr)
    {
        return E_POINTER;
    }

    if (shutdown_)
    {
        return MF_E_SHUTDOWN;
    }

    if (streamDescriptor_ != nullptr)
    {
        streamDescriptor_->Release();

        streamDescriptor_ = nullptr;
    }

    streamDescriptor_ =
        descriptor;

    streamDescriptor_->AddRef();

    std::cout
        << "[LensLinksMediaStream] "
        << "stream descriptor assigned"
        << std::endl;

    return S_OK;
}
